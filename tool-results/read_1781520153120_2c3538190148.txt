   103→function getCheapestPrice(product: ProductCard): string {
   104→  try {
   105→    const opts = JSON.parse(product.priceOptions || '[]')
   106→    if (Array.isArray(opts) && opts.length > 0) {
   107→      const sorted = [...opts].sort((a: Record<string, unknown>, b: Record<string, unknown>) => {
   108→        const priceA = parseInt(String(a.priceBDT).replace(/\D/g, '') || '0')
   109→        const priceB = parseInt(String(b.priceBDT).replace(/\D/g, '') || '0')
   110→        return priceA - priceB
   111→      })
   112→      return String(sorted[0].priceBDT)
   113→    }
   114→  } catch {
   115→    // fallback to base price
   116→  }
   117→  return product.basePriceBDT
   118→}

   122→function detectLanguageClient(message: string): DetectedLang {
   123→  if (/[\u0980-\u09FF]/.test(message)) return 'bangla'
   124→  const patterns = [/koto/i, /taka/i, /lagbe/i, /chai/i, /order\s*korbo/i, /nite\s*chai/i, /kinte\s*chai/i, /dekhao/i, /ki\s*ki/i, /sob/i, /nam/i, /amar/i, /apnar/i, /bhai/i, /kichu/i, /kemon/i, /valo/i, /ase/i, /ache/i, /korbo/i, /parbo/i, /kivabe/i, /keno/i]
   125→  if (patterns.filter((p) => p.test(message)).length >= 2) return 'banglish'
   126→  return 'english'
   127→}

   131→// Chrome loads voices asynchronously — this helper ensures they're available
   132→function getVoicesAsync(): Promise<SpeechSynthesisVoice[]> {
   133→  return new Promise((resolve) => {
   134→    if (typeof window === 'undefined' || !window.speechSynthesis) {
   135→      resolve([])
   136→      return
   137→    }
   138→    const voices = window.speechSynthesis.getVoices()
   139→    if (voices.length > 0) {
   140→      resolve(voices)
   141→      return
   142→    }
   143→    // Voices not loaded yet — wait for voiceschanged event
   144→    const onVoicesChanged = () => {
   145→      const v = window.speechSynthesis.getVoices()
   146→      if (v.length > 0) {
   147→        window.speechSynthesis.removeEventListener('voiceschanged', onVoicesChanged)
   148→        resolve(v)
   149→      }
   150→    }
   151→    window.speechSynthesis.addEventListener('voiceschanged', onVoicesChanged)
   152→    // Safety: resolve after 2s even if no voices loaded
   153→    setTimeout(() => {
   154→      window.speechSynthesis.removeEventListener('voiceschanged', onVoicesChanged)
   155→      resolve(window.speechSynthesis.getVoices())
   156→    }, 2000)
   157→  })
   158→}

   160→// Chrome workaround: speechSynthesis pauses after ~15s of inactivity.
   161→// This keeps it alive by calling resume() periodically while speaking.
   162→let chromeKeepAliveInterval: ReturnType<typeof setInterval> | null = null
   163→function startChromeKeepAlive() {
   164→  if (chromeKeepAliveInterval) return
   165→  chromeKeepAliveInterval = setInterval(() => {
   166→    if (window.speechSynthesis?.speaking) {
   167→      window.speechSynthesis.resume()
   168→    } else {
   169→      stopChromeKeepAlive()
   170→    }
   171→  }, 10000)
   172→}

   174→  if (chromeKeepAliveInterval) {
   175→    clearInterval(chromeKeepAliveInterval)
   176→    chromeKeepAliveInterval = null
   177→  }
   178→}

   184→async function speakWithServerTTS(text: string, lang: DetectedLang, onEnd?: () => void): Promise<boolean> {
   185→  // Clean text for TTS on the client side too (for the request body)
   186→  const cleanText = text
   187→    .replace(/[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F1E0}-\u{1F1FF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}\u{FE00}-\u{FE0F}\u{1F900}-\u{1F9FF}\u{200D}\u{20E3}\u{E0020}-\u{E007F}]/gu, '')
   188→    .replace(/\*\*(.+?)\*\*/g, '$1')
   189→    .replace(/\[(.+?)\]\((.+?)\)/g, '$1')
   190→    .replace(/[•\-]\s/g, '')
   191→    .replace(/\d+[.)]\s/g, '')
   192→    .replace(/৳/g, 'টাকা')
   193→    .replace(/BDT/gi, 'টাকা')
   194→    .replace(/\bbKash\b/gi, 'বিকাশ')
   195→    .replace(/\bNagad\b/gi, 'নগদ')
   196→    .replace(/\bWhatsApp\b/gi, 'হোয়াটসঅ্যাপ')
   197→    .replace(/↵/g, ' ')
   198→    .replace(/\n/g, ' ')
   199→    .trim()
   200→
   201→  if (!cleanText) { onEnd?.(); return true }
   202→
   203→  try {
   204→    // Stop any currently playing audio
   205→    if (currentAudio) {
   206→      currentAudio.pause()
   207→      currentAudio = null
   208→    }
   209→
   210→    const res = await fetch('/api/tts', {
   211→      method: 'POST',
   212→      headers: { 'Content-Type': 'application/json' },
   213→      body: JSON.stringify({ text: cleanText, lang }),
   214→    })
   215→
   216→    // If server TTS returns 503 with fallback flag, use browser TTS
   217→    if (res.status === 503) {
   218→      console.warn('Server TTS unavailable (ElevenLabs + OpenAI failed), falling back to browser TTS')
   219→      return false
   220→    }
   221→
   222→    if (!res.ok) {
   223→      console.warn('Server TTS failed, falling back to browser TTS')
   224→      return false
   225→    }
   226→
   227→    const audioBlob = await res.blob()
   228→    const audioUrl = URL.createObjectURL(audioBlob)
   229→    const audio = new Audio(audioUrl)
   230→    currentAudio = audio
   231→
   232→    return new Promise<boolean>((resolve) => {
   233→      const safetyTimer = setTimeout(() => {
   234→        audio.pause()
   235→        currentAudio = null
   236→        URL.revokeObjectURL(audioUrl)
   237→        onEnd?.()
   238→        resolve(true)
   239→      }, 45000)
   240→
   241→      audio.onended = () => {
   242→        clearTimeout(safetyTimer)
   243→        currentAudio = null
   244→        URL.revokeObjectURL(audioUrl)
   245→        onEnd?.()
   246→        resolve(true)
   247→      }
   248→
   249→      audio.onerror = () => {
   250→        clearTimeout(safetyTimer)
   251→        currentAudio = null
   252→        URL.revokeObjectURL(audioUrl)
   253→        console.warn('Audio playback error, falling back to browser TTS')
   254→        resolve(false)
   255→      }
   256→
   257→      audio.play().catch(() => {
   258→        clearTimeout(safetyTimer)
   259→        currentAudio = null
   260→        URL.revokeObjectURL(audioUrl)
   261→        resolve(false)
   262→      })
   263→    })
   264→  } catch {
   265→    console.warn('Server TTS error, falling back to browser TTS')
   266→    return false
   267→  }
   268→}
   269→
   270→function stopServerAudio() {
   271→  if (currentAudio) {
   272→    currentAudio.pause()
   273→    currentAudio = null
   274→  }
   275→}

   277→async function speak(text: string, lang: DetectedLang, onEnd?: () => void): Promise<SpeechSynthesisUtterance | null> {
   278→  if (typeof window === 'undefined') return null
   279→
   280→  // Stop any existing audio
   281→  stopServerAudio()
   282→  if (window.speechSynthesis) window.speechSynthesis.cancel()
   283→  stopChromeKeepAlive()
   284→
   285→  // Try Server TTS first (ElevenLabs → OpenAI → Browser fallback)
   286→  const serverSuccess = await speakWithServerTTS(text, lang, onEnd)
   287→  if (serverSuccess) return null // Server TTS handled it
   288→
   289→  // Fallback to browser TTS
   290→  if (!window.speechSynthesis) { onEnd?.(); return null }
   291→
   292→  // Clean text for speech
   293→  const cleanText = text
   294→    .replace(/[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F1E0}-\u{1F1FF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}\u{FE00}-\u{FE0F}\u{1F900}-\u{1F9FF}\u{200D}\u{20E3}\u{E0020}-\u{E007F}]/gu, '')
   295→    .replace(/\*\*(.+?)\*\*/g, '$1')
   296→    .replace(/\[(.+?)\]\((.+?)\)/g, '$1')
   297→    .replace(/[•\-]\s/g, '')
   298→    .replace(/\d+[.)]\s/g, '')
   299→    .replace(/৳/g, 'টাকা')
   300→    .replace(/BDT/gi, 'টাকা')
   301→    .replace(/\bbKash\b/gi, 'বিকাশ')
   302→    .replace(/\bNagad\b/gi, 'নগদ')
   303→    .replace(/\bWhatsApp\b/gi, 'হোয়াটসঅ্যাপ')
   304→    .replace(/↵/g, ' ')
   305→    .replace(/\n/g, ' ')
   306→    .trim()
   307→
   308→  if (!cleanText) { onEnd?.(); return null }
   309→
   310→  const voices = await getVoicesAsync()
   311→  const utterance = new SpeechSynthesisUtterance(cleanText)
   312→
   313→  if (lang === 'bangla' || lang === 'banglish') {
   314→    const bnVoice = voices.find(v => v.lang === 'bn-BD') || voices.find(v => v.lang === 'bn-IN')
   315→    if (bnVoice) {
   316→      utterance.voice = bnVoice
   317→      utterance.lang = bnVoice.lang
   318→    } else {
   319→      const hiVoice = voices.find(v => v.lang === 'hi-IN')
   320→      if (hiVoice) { utterance.voice = hiVoice; utterance.lang = 'hi-IN' }
   321→      else { utterance.lang = 'bn-BD' }
   322→    }
   323→  } else {
   324→    utterance.lang = 'en-US'
   325→    const enVoice = voices.find(v => v.lang === 'en-US' && v.name.toLowerCase().includes('female'))
   326→      || voices.find(v => v.lang === 'en-US')
   327→    if (enVoice) utterance.voice = enVoice
   328→  }
   329→
   330→  utterance.rate = 0.9
   331→  utterance.pitch = 1.15
   332→  utterance.volume = 1.0
   333→
   334→  startChromeKeepAlive()
   335→
   336→  const safetyTimer = setTimeout(() => {
   337→    if (window.speechSynthesis?.speaking) window.speechSynthesis.cancel()
   338→    stopChromeKeepAlive()
   339→    onEnd?.()
   340→  }, 45000)
   341→
   342→  const wrappedOnEnd = () => {
   343→    clearTimeout(safetyTimer)
   344→    stopChromeKeepAlive()
   345→    onEnd?.()
   346→  }
   347→
   348→  utterance.onend = wrappedOnEnd
   349→  utterance.onerror = () => wrappedOnEnd()
   350→
   351→  await new Promise(resolve => setTimeout(resolve, 100))
   352→  window.speechSynthesis.speak(utterance)
   353→  return utterance
   354→}

   358→function formatMessage(text: string): ReactNode[] {
   359→  const lines = text.split('\n')
   360→  const elements: ReactNode[] = []
   361→
   362→  lines.forEach((line, lineIdx) => {
   363→    const processedLine = processInlineFormatting(line)
   364→
   365→    if (lineIdx < lines.length - 1) {
   366→      elements.push(
   367→        <span key={`line-${lineIdx}`}>
   368→          {processedLine}
   369→          <br />
   370→        </span>
   371→      )
   372→    } else {
   373→      elements.push(<span key={`line-${lineIdx}`}>{processedLine}</span>)
   374→    }
   375→  })
   376→
   377→  return elements
   378→}
   379→
   380→function processInlineFormatting(line: string): ReactNode[] {
   381→  const elements: ReactNode[] = []
   382→  let remaining = line
   383→
   384→  // Check if this is a bullet point line (• or -)
   385→  const bulletMatch = remaining.match(/^(\s*)([•\-])\s(.*)/)
   386→  if (bulletMatch) {
   387→    const [, indent, , content] = bulletMatch
   388→    elements.push(
   389→      <span key={`bullet-${indent.length}`} className="flex items-start gap-1.5 my-0.5">
   390→        <span className="mt-[7px] shrink-0 h-1.5 w-1.5 rounded-full bg-primary" />
   391→        <span>{processInlineFormatting(content)}</span>
   392→      </span>
   393→    )
   394→    return elements
   395→  }
   396→
   397→  // Check if this is a numbered list item (1. 2. etc.)
   398→  const numberedMatch = remaining.match(/^(\s*)(\d+)[.)]\s(.*)/)
   399→  if (numberedMatch) {
   400→    const [, , num, content] = numberedMatch
   401→    elements.push(
   402→      <span key={`num-${num}`} className="flex items-start gap-1.5 my-0.5">
   403→        <span className="shrink-0 min-w-[18px] h-[18px] rounded-full bg-primary/15 text-primary flex items-center justify-center text-[10px] font-bold leading-none mt-px">
   404→          {num}
   405→        </span>
   406→        <span>{processInlineFormatting(content)}</span>
   407→      </span>
   408→    )
   409→    return elements
   410→  }
   411→
   412→  // Process inline: **bold** and links
   413→  let keyIdx = 0
   414→  while (remaining.length > 0) {
   415→    // Bold pattern: **text**
   416→    const boldMatch = remaining.match(/\*\*(.+?)\*\*/)
   417→    if (boldMatch && boldMatch.index !== undefined) {
   418→      if (boldMatch.index > 0) {
   419→        elements.push(<span key={`t-${keyIdx++}`}>{remaining.slice(0, boldMatch.index)}</span>)
   420→      }
   421→      elements.push(
   422→        <strong key={`b-${keyIdx++}`} className="font-semibold text-teal-700">
   423→          {boldMatch[1]}
   424→        </strong>
   425→      )
   426→      remaining = remaining.slice(boldMatch.index + boldMatch[0].length)
   427→      continue
   428→    }
   429→
   430→    // Link pattern: [text](url)
   431→    const linkMatch = remaining.match(/\[(.+?)\]\((.+?)\)/)
   432→    if (linkMatch && linkMatch.index !== undefined) {
   433→      if (linkMatch.index > 0) {
   434→        elements.push(<span key={`t-${keyIdx++}`}>{remaining.slice(0, linkMatch.index)}</span>)
   435→      }
   436→      elements.push(
   437→        <a
   438→          key={`a-${keyIdx++}`}
   439→          href={linkMatch[2]}
   440→          target="_blank"
   441→          rel="noopener noreferrer"
   442→          className="text-teal-600 underline underline-offset-2 hover:text-teal-800 transition-colors"
   443→        >
   444→          {linkMatch[1]}
   445→        </a>
   446→      )
   447→      remaining = remaining.slice(linkMatch.index + linkMatch[0].length)
   448→      continue
   449→    }
   450→
   451→    // No more matches, push the rest
   452→    elements.push(<span key={`t-${keyIdx++}`}>{remaining}</span>)
   453→    break
   454→  }
   455→
   456→  return elements
   457→}

   528→function SpeakerButton({ text, lang, isStreaming }: { text: string; lang: DetectedLang; isStreaming?: boolean }) {
   529→  const [isSpeaking, setIsSpeaking] = useState(false)
   530→
   531→  const handleSpeak = useCallback(() => {
   532→    if (isSpeaking) {
   533→      stopServerAudio()
   534→      if (typeof window !== 'undefined' && window.speechSynthesis) {
   535→        window.speechSynthesis.cancel()
   536→      }
   537→      setIsSpeaking(false)
   538→      return
   539→    }
   540→
   541→    if (isStreaming || !text) return
   542→
   543→    setIsSpeaking(true)
   544→    speak(text, lang, () => {
   545→      setIsSpeaking(false)
   546→    })
   547→
   548→    // Safety: auto-reset after max 30 seconds
   549→    setTimeout(() => setIsSpeaking(false), 30000)
   550→  }, [text, lang, isSpeaking, isStreaming])
   551→
   552→  // Clean up on unmount
   553→  useEffect(() => {
   554→    return () => {
   555→      if (typeof window !== 'undefined' && window.speechSynthesis) {
   556→        window.speechSynthesis.cancel()
   557→      }
   558→    }
   559→  }, [])
   560→
   561→  if (isStreaming) return null
   562→
   563→  return (
   564→    <button
   565→      onClick={handleSpeak}
   566→      className={`inline-flex items-center justify-center h-6 w-6 rounded-full transition-all active:scale-90 touch-manipulation ${
   567→        isSpeaking
   568→          ? 'bg-[#00A6A6] text-white shadow-md shadow-[#00A6A6]/30 animate-pulse'
   569→          : 'bg-muted/60 hover:bg-[#00A6A6]/15 text-muted-foreground hover:text-[#00A6A6]'
   570→      }`}
   571→      aria-label={isSpeaking ? 'বন্ধ করুন' : 'শুনুন'}
   572→      title={isSpeaking ? 'Stop' : 'Listen'}
   573→    >
   574→      {isSpeaking ? (
   575→        <VolumeX className="h-3 w-3" />
   576→      ) : (
   577→        <Volume2 className="h-3 w-3" />
   578→      )}
   579→    </button>
   580→  )
   581→}

   630→  const scrollRef = useRef<HTMLDivElement>(null)

   634→
   635→  // ── Typewriter effect for the floating bubble ──

   687→  const sendMessage = useCallback(async (overrideMessage?: string) => {
   688→    const trimmed = (overrideMessage || input).trim()
   689→    if (!trimmed || isLoadingRef.current || cooldown) return
   690→
   691→    // Rate limit: minimum 1.5s between messages
   692→    const now = Date.now()
   693→    const elapsed = now - lastSentRef.current
   694→    if (elapsed < 1500) {
   695→      setCooldown(true)
   696→      setTimeout(() => setCooldown(false), 1500 - elapsed)
   697→      return
   698→    }
   699→    lastSentRef.current = now
   700→
   701→    const userMessage: ChatMessage = {
   702→      role: 'user',
   703→      content: trimmed,
   704→      timestamp: Date.now(),
   705→    }
   706→    setMessages(prev => [...prev, userMessage])
   707→    if (!overrideMessage) setInput('')
   708→    setIsLoading(true)
   709→
   710→    // Detect language from user message for TTS matching later
   711→    const userDetectedLang = detectLanguageClient(trimmed)
   712→
   713→    // Add empty streaming assistant message
   714→    const assistantTimestamp = Date.now()
   715→    setMessages(prev => [
   716→      ...prev,
   717→      { role: 'assistant', content: '', timestamp: assistantTimestamp, isStreaming: true, detectedLang: userDetectedLang },
   718→    ])
   719→

   721→      // Use messagesRef to avoid stale closure and unnecessary re-creates
   722→      const currentMessages = messagesRef.current
   723→      const history = currentMessages.slice(-10).map(m => ({
   724→        role: m.role,
   725→        content: m.content,

   732→          message: trimmed,

   737→

   740→        setMessages(prev => {
   741→          const updated = [...prev]
   742→          const lastIdx = updated.length - 1
   743→          if (updated[lastIdx]?.role === 'assistant') {
   744→            updated[lastIdx] = {
   745→              ...updated[lastIdx],
   746→              content: '⏳ একটু ব্যস্ত আছি, কিছুক্ষণ পর আবার চেষ্টা করুন। অথবা সরাসরি WhatsApp এ যোগাযোগ করুন: +8801647236359 💬',
   747→              whatsappUrl: 'https://wa.me/8801647236359?text=' + encodeURIComponent('Hi, Streaming Hub এ সাহায্য দরকার'),
   748→              isStreaming: false,
   749→            }

   752→        })

   754→        return

   763→      let buffer = ''
   764→
   765→      while (true) {
   766→        const { done, value } = await reader.read()
   767→        if (done) break
   768→
   769→        buffer += decoder.decode(value, { stream: true })
   770→        const lines = buffer.split('\n')
   771→        buffer = lines.pop() || ''
   772→

   796→                }
   797→                return updated
   798→              })

   800→              setMessages(prev => {
   801→                const updated = [...prev]
   802→                const lastIdx = updated.length - 1
   803→                if (updated[lastIdx]?.role === 'assistant') {
   804→                  updated[lastIdx] = { ...updated[lastIdx], suggestions: data.suggestions }
   805→                }
   806→                return updated
   807→              })
   808→            } else if (data.type === 'done') {
   809→              setMessages(prev => {
   810→                const updated = [...prev]
   811→                const lastIdx = updated.length - 1
   812→                if (updated[lastIdx]?.role === 'assistant') {

   826→            // ignore parse errors for malformed SSE events
   827→          }
   828→        }

   832→      setMessages(prev => {

   836→          updated[lastIdx] = { ...updated[lastIdx], isStreaming: false }
   837→        }
   838→        return updated
   839→      })
   840→
   841→      // NOTE: Voice mode TTS is now handled by the useEffect below,
   842→      // which watches for the last assistant message's isStreaming → false transition.
   843→      // This avoids the stale closure problem with the old setTimeout approach.
   844→    } catch {
   845→      setMessages(prev => {
   846→        const updated = [...prev]
   847→        const lastIdx = updated.length - 1
   848→        if (updated[lastIdx]?.role === 'assistant') {
   849→          updated[lastIdx] = {
   850→            ...updated[lastIdx],
   851→            content: '❌ সংযোগে সমস্যা হচ্ছে। আবার চেষ্টা করুন অথবা WhatsApp এ যোগাযোগ করুন: +8801647236359',
   852→            isStreaming: false,
   853→          }
   854→        }
   855→        return updated
   856→      })
   857→    } finally {
   858→      setIsLoading(false)
   859→    }
   860→  }, [input, cooldown])
   861→

   863→  useEffect(() => {
   864→    sendMessageRef.current = sendMessage
   865→  }, [sendMessage])
   866→
   867→  // ── Voice Loop: TTS after AI finishes, then auto-listen ──
   868→  // This useEffect watches for the last assistant message's isStreaming
   869→  // transitioning to false, and triggers TTS → auto-listen in voice mode.
   870→  // Using refs avoids stale closure issues.
   871→  const voiceLoopTriggeredRef = useRef<number>(0) // timestamp of last triggered TTS
   872→  useEffect(() => {
   873→    if (!isVoiceModeRef.current || !speechSupported) return
   874→
   875→    const lastMsg = messages[messages.length - 1]
   876→    if (!lastMsg || lastMsg.role !== 'assistant' || lastMsg.isStreaming || !lastMsg.content) return
   877→
   878→    // Avoid re-triggering for the same message
   879→    if (voiceLoopTriggeredRef.current === lastMsg.timestamp) return
   880→    voiceLoopTriggeredRef.current = lastMsg.timestamp
   881→
   882→    // Skip speaking the initial greeting when first entering voice mode
   883→    // (user hasn't asked anything yet)
   884→    const isInitialGreeting = messages.length <= 1 && lastMsg.content === KORMOCHARY_GREETING
   885→    if (isInitialGreeting) {
   886→      // Just start listening immediately
   887→      const rec = recognitionRef.current
   888→      if (rec && !isAISpeakingRef.current && !isLoadingRef.current) {
   889→        setTimeout(() => {
   890→          try {
   891→            rec.start()
   892→            setIsListening(true)
   893→          } catch {
   894→            // Recognition may already be started
   895→          }
   896→        }, 300)
   897→      }
   898→      return
   899→    }
   900→
   901→    // Small delay to ensure the state has settled after isStreaming → false
   902→    const timer = setTimeout(() => {
   903→      if (!isVoiceModeRef.current) return // Voice mode was disabled during delay
   904→
   905→      setIsAISpeaking(true)
   906→      const lang = lastMsg.detectedLang || 'bangla'
   907→
   908→      speak(lastMsg.content, lang, () => {
   909→        setIsAISpeaking(false)
   910→        // After AI finishes speaking, auto-start listening again if still in voice mode
   911→        if (!isVoiceModeRef.current) return
   912→        const rec = recognitionRef.current
   913→        if (rec) {
   914→          // Small delay before starting recognition again to avoid race condition
   915→          setTimeout(() => {
   916→            if (!isVoiceModeRef.current || isAISpeakingRef.current || isLoadingRef.current) return
   917→            try {
   918→              rec.start()
   919→              setIsListening(true)
   920→            } catch {
   921→              // Recognition may already be started or not available
   922→            }
   923→          }, 500)
   924→        }
   925→      })
   926→    }, 500)
   927→
   928→    return () => clearTimeout(timer)
   929→  }, [messages, speechSupported])
   930→
   931→  // ── Initialize Speech Recognition (Dynamic Language STT) ──
   932→  useEffect(() => {
   933→    if (!speechSupported) return
   934→    const SR = (window as unknown as Record<string, unknown>)['SpeechRecognition'] ||
   935→      (window as unknown as Record<string, unknown>)['webkitSpeechRecognition']
   936→    if (!SR) return
   937→
   938→    const recognition = new (SR as new () => SpeechRecognition)()
   939→    recognition.lang = 'bn-BD'
   940→    recognition.continuous = false
   941→    recognition.interimResults = true
   942→
   943→    recognition.onresult = (event: SpeechRecognitionEvent) => {
   944→      const transcript = event.results[0][0].transcript
   945→      setInput(transcript)
   946→
   947→      if (event.results[0].isFinal) {
   948→        setIsListening(false)
   949→
   950→        const spokenLang = detectLanguageClient(transcript)
   951→
   952→        // Update recognition language for next session
   953→        try {
   954→          if (spokenLang === 'english') {
   955→            recognition.lang = 'en-US'
   956→          } else {
   957→            recognition.lang = 'bn-BD'
   958→          }
   959→        } catch {
   960→          // Can't change lang while recognition might be active
   961→        }
   962→
   963→        // Auto-send after final result (in both voice mode and regular mic)
   964→        setTimeout(() => {
   965→          sendMessageRef.current?.(transcript)
   966→        }, 150)
   967→      }
   968→    }
   969→
   970→    recognition.onerror = (event) => {
   971→      setIsListening(false)
   972→      // In voice mode, if recognition errors (e.g., no-speech), try to restart
   973→      // Common errors: no-speech (user was quiet), aborted (manual stop), not-allowed (mic denied)
   974→      if (isVoiceModeRef.current && event.error !== 'aborted' && event.error !== 'not-allowed') {
   975→        setTimeout(() => {
   976→          if (isVoiceModeRef.current && !isAISpeakingRef.current && !isLoadingRef.current) {
   977→            try {
   978→              recognition.start()
   979→              setIsListening(true)
   980→            } catch {
   981→              // Recognition may already be started
   982→            }
   983→          }
   984→        }, 800) // Longer delay to avoid rapid restart loops
   985→      }
   986→      if (event.error === 'not-allowed') {
   987→        console.warn('Microphone access denied. Voice mode requires microphone permission.')
   988→        // Auto-exit voice mode if mic is denied
   989→        setIsVoiceMode(false)
   990→        isVoiceModeRef.current = false
   991→      }
   992→    }
   993→
   994→    recognition.onend = () => {
   995→      setIsListening(false)
   996→      // In voice mode, auto-restart listening after recognition ends naturally
   997→      // (but only if AI is not currently speaking/loading)
   998→      if (isVoiceModeRef.current && !isAISpeakingRef.current && !isLoadingRef.current) {
   999→        setTimeout(() => {
  1000→          if (isVoiceModeRef.current && !isAISpeakingRef.current && !isLoadingRef.current) {
  1001→            try {
  1002→              recognition.start()
  1003→              setIsListening(true)
  1004→            } catch {
  1005→              // Recognition may already be started
  1006→            }
  1007→          }
  1008→        }, 300)
  1009→      }
  1010→    }
  1011→
  1012→    recognitionRef.current = recognition
  1013→  }, [speechSupported])
  1014→
  1015→  const toggleListening = useCallback(() => {
  1016→    const recognition = recognitionRef.current
  1017→    if (!recognition) return
  1018→    if (isListening) {
  1019→      recognition.stop()
  1020→      setIsListening(false)
  1021→    } else {
  1022→      try {
  1023→        if (input.trim()) {
  1024→          const inputLang = detectLanguageClient(input)
  1025→          recognition.lang = inputLang === 'english' ? 'en-US' : 'bn-BD'
  1026→        }
  1027→        recognition.start()
  1028→        setIsListening(true)
  1029→      } catch {
  1030→        setIsListening(false)
  1031→      }
  1032→    }
  1033→  }, [isListening, input])
  1034→
  1035→  // ── Voice Mode Toggle ──
  1036→  const toggleVoiceMode = useCallback(() => {
  1037→    if (isVoiceMode) {
  1038→      // Exit voice mode
  1039→      setIsVoiceMode(false)
  1040→      isVoiceModeRef.current = false
  1041→      setIsAISpeaking(false)
  1042→      isAISpeakingRef.current = false
  1043→      stopServerAudio()
  1044→      if (typeof window !== 'undefined' && window.speechSynthesis) {
  1045→        window.speechSynthesis.cancel()
  1046→      }
  1047→      if (isListening && recognitionRef.current) {
  1048→        recognitionRef.current.stop()
  1049→        setIsListening(false)
  1050→      }
  1051→    } else {
  1052→      // Enter voice mode
  1053→      setIsVoiceMode(true)
  1054→      isVoiceModeRef.current = true
  1055→      // Start listening immediately
  1056→      const recognition = recognitionRef.current
  1057→      if (recognition) {
  1058→        try {
  1059→          recognition.start()
  1060→          setIsListening(true)
  1061→        } catch {
  1062→          setIsListening(false)
  1063→        }
  1064→      }
  1065→    }
  1066→  }, [isVoiceMode, isListening])
  1067→
  1068→  const handleKeyDown = (e: React.KeyboardEvent) => {
  1069→    if (e.key === 'Enter' && !e.shiftKey) {
  1070→      e.preventDefault()
  1071→      sendMessage()
  1072→    }
  1073→  }
  1074→
