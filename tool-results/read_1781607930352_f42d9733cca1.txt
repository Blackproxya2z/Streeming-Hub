   171→function buildProductContext(intent: Intent, userMsg: string): string {
   172→  const parts: string[] = []
   173→  const addProduct = (p: Product) => {
   174→    parts.push(`Name: ${safeName(p)} | Category: ${p.category.name}`)
   175→    parts.push(`Price Options:\n${parsePriceOptionsDetailed(p)}`)
   176→    parts.push(`Cheapest: ${getCheapestPlan(p)} | Stock: ${p.stockStatus} | Warranty: ${p.warranty || 'Full warranty'} | Delivery: ${p.deliveryTime}`)
   177→  }
   178→
   179→  switch (intent) {
   180→    case 'pin_inquiry': case 'greeting': case 'thanks': case 'goodbye':
   181→      parts.push('=== CATALOG OVERVIEW ===', briefSummary()); break
   182→    case 'specific_product': {
   183→      const p = findSpecificProduct(userMsg)
   184→      if (p) { parts.push('=== SPECIFIC PRODUCT ==='); addProduct(p); const f = parseFeatures(p); if (f.length) parts.push(`Features: ${f.slice(0, 8).join(', ')}`); const rel = findRelatedProducts(p.id, 3); if (rel.length) { parts.push('\n=== RELATED ==='); rel.forEach((r) => parts.push(`• ${safeName(r)} — From ${getCheapestPlan(r)}`)) } }
   185→      break
   186→    }
   187→    case 'category': {
   188→      const slug = detectCategorySlug(userMsg)
   189→      if (slug) { const { category, products, totalCount } = searchByCategory(slug); if (category && products.length) { parts.push(`=== CATEGORY: ${category.isAdult ? sanitizeText(category.name) : category.name} (${totalCount}) ===`); products.slice(0, 8).forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)}${p.isBestSeller ? ' 🔥' : p.isFeatured ? ' ⭐' : ''} — From ${getCheapestPlan(p)}`)) } }
   190→      break
   191→    }
   192→    case 'price_inquiry': case 'search': {
   193→      const results = searchProducts(userMsg, 6)
   194→      if (results.length) { parts.push('=== SEARCH RESULTS ==='); results.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} — From ${getCheapestPlan(p)} | ${p.category.name} | ${p.stockStatus}`)) }
   195→      const spec = findSpecificProduct(userMsg)
   196→      if (spec) { parts.push('\n=== EXACT MATCH ==='); addProduct(spec) }
   197→      break
   198→    }
   199→    case 'featured': {
   200→      const { products: feat } = getFeaturedProducts(6)
   201→      if (feat.length) { parts.push('=== FEATURED / BEST SELLERS ==='); feat.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} ${p.isBestSeller ? '🔥' : p.isNewArrival ? '✨' : ''} — From ${getCheapestPlan(p)}`)) }
   202→      break
   203→    }
   204→    case 'all_products': {
   205→      parts.push('=== FULL CATALOG ==='); getCatalogSummary().forEach((c) => parts.push(`📂 ${c.isAdult ? sanitizeText(c.name) : c.name} (${c.productCount}) — slug: ${c.slug}`, `   Popular: ${c.sampleProducts.slice(0, 3).join(', ')}`)); break
   206→    }
   207→    case 'comparison': {
   208→      const results = searchProducts(userMsg, 4)
   209→      if (results.length) { parts.push('=== COMPARISON DATA ==='); results.forEach((p) => { parts.push(`\n${safeName(p)}:`); parts.push(`  Price:\n${parsePriceOptionsDetailed(p)}`); parts.push(`  Cheapest: ${getCheapestPlan(p)}`); const f = parseFeatures(p); if (f.length) parts.push(`  Features: ${f.slice(0, 6).join(', ')}`); parts.push(`  Stock: ${p.stockStatus} | Warranty: ${p.warranty || 'Full'}`) }) }
   210→      break
   211→    }
   212→    case 'order_payment': case 'how_to_use': case 'warranty_delivery': {
   213→      const p = findSpecificProduct(userMsg)
   214→      if (p) { parts.push('=== RELEVANT PRODUCT ==='); addProduct(p) } else { parts.push('=== CATALOG OVERVIEW ===', briefSummary()) }
   215→      break
   216→    }
   217→    case 'out_of_scope': {
   218→      const { products: feat } = getFeaturedProducts(3)
   219→      if (feat.length) { parts.push('=== TOP PICKS ==='); feat.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} — From ${getCheapestPlan(p)}`)) }
   220→      break

   228→function collectProducts(intent: Intent, userMsg: string, hasRestrictedAccess: boolean): ProductCard[] {
   229→  const products: Product[] = []
   230→  switch (intent) {
   231→    case 'specific_product': { const p = findSpecificProduct(userMsg); if (p) { products.push(p); products.push(...findRelatedProducts(p.id, 3)) } break }
   232→    case 'category': { const slug = detectCategorySlug(userMsg); if (slug) products.push(...searchByCategory(slug).products.slice(0, 8)); break }
   233→    case 'price_inquiry': case 'search': products.push(...searchProducts(userMsg, 6)); break
   234→    case 'featured': products.push(...getFeaturedProducts(6).products); break
   235→    case 'all_products': getCatalogSummary().filter((c) => !c.isAdult).forEach((c) => products.push(...searchByCategory(c.slug).products.slice(0, 2))); break
   236→    case 'comparison': products.push(...searchProducts(userMsg, 4)); break
   237→    case 'out_of_scope': products.push(...getFeaturedProducts(3).products); break
   238→    case 'order_payment': case 'how_to_use': case 'warranty_delivery': { const p = findSpecificProduct(userMsg); if (p) products.push(p); break }
   239→    default: break
   240→  }
   241→  const seen = new Set<string>()
   242→  return products
   243→    .filter((p) => !seen.has(p.id) && seen.add(p.id))
   244→    .filter((p) => hasRestrictedAccess || !isRestrictedProduct(p))
   245→    .map(toProductCard)
   246→}

   459→function sanitizeHistory(history: ChatMessage[]): ChatMessage[] {
   460→  const allowedRoles = new Set(['system', 'user', 'assistant'])
   461→  return history
   462→    .filter((m) => allowedRoles.has(m.role) && m.content && m.content.trim().length > 0)

   466→// ─── POST Handler (SSE Streaming with OpenAI) ──────────────────────────────
   467→
   468→export async function POST(request: NextRequest) {
   469→  let body: { message?: string; history?: ChatMessage[]; sessionId?: string }
   470→  try { body = await request.json() } catch { return new Response(JSON.stringify({ error: 'Invalid JSON' }), { status: 400 }) }
   471→  const { message, history = [], sessionId = 'default' } = body
   472→  if (!message || typeof message !== 'string') return new Response(JSON.stringify({ error: 'Message required' }), { status: 400 })
   473→  if (!checkRateLimit(sessionId)) return new Response(JSON.stringify({ error: 'Rate limit exceeded' }), { status: 429 })
   474→
   475→  const intent = detectIntent(message, history)
   476→  const lang = detectLanguage(message)
   477→  const restrictedAccess = hasRestrictedAccessFromRequest(request)
   478→  const products = collectProducts(intent, message, restrictedAccess)
   479→  const suggestions = generateSuggestions(intent, lang)
   480→  const whatsappUrl = buildWhatsAppUrl({})
   481→  const systemPrompt = buildSystemPrompt(intent, lang, message)
   482→
   483→  // Sanitize history: only allow system/user/assistant roles, latest 8-10 messages, remove empty content
   484→  const cleanHistory = sanitizeHistory(history)
   485→  const messages: OpenAI.ChatCompletionMessageParam[] = [
   486→    { role: 'system', content: systemPrompt },
   487→    ...cleanHistory.map((m) => ({ role: m.role as 'user' | 'assistant', content: m.content })),
   488→    { role: 'user', content: message },
   489→  ]
   490→
   491→  const stream = new ReadableStream({
   492→    async start(controller) {
   493→      const enc = new TextEncoder()
   494→      const send = (data: Record<string, unknown>) => controller.enqueue(enc.encode(`data: ${JSON.stringify(data)}\n\n`))
   495→
   496→      // All LLM calls are wrapped with an 8-second timeout to prevent
   497→      // long hangs when the external API is unreachable.
   498→      const LLM_TIMEOUT_MS = 8000
   499→      let llmContent = ''
   500→
   501→      try {
   502→        // Try streaming with OpenAI (with timeout)
   503→        try {
   504→          const completion = await withTimeout(
   505→            getOpenAI().chat.completions.create({
   506→              model: 'gpt-4o-mini',
   507→              messages,
   508→              stream: true,
   509→            }),
   510→            LLM_TIMEOUT_MS,
   511→          )
   512→
   513→          // Use for-await-of loop over the async iterable stream
   514→          for await (const chunk of completion) {
   515→            const delta = chunk.choices?.[0]?.delta?.content
   516→            if (delta) {
   517→              send({ type: 'token', content: delta })
   518→              llmContent += delta
   519→            }
   520→          }
   521→        } catch (streamErr) {
   522→          // If streaming timed out, the API is unreachable — skip non-streaming attempt
   523→          // and go straight to fallback (avoids another wait)
   524→          if (streamErr instanceof Error && streamErr.message === 'LLM_TIMEOUT') {
   525→            send({ type: 'token', content: getFallback(intent, lang) })
   526→          } else {
   527→            // Non-timeout error — try non-streaming as fallback
   528→            try {
   529→              const completion = await withTimeout(
   530→                getOpenAI().chat.completions.create({
   531→                  model: 'gpt-4o-mini',
   532→                  messages,
   533→                  stream: false,
   534→                }),
   535→                LLM_TIMEOUT_MS,
   536→              )
   537→              llmContent = completion?.choices?.[0]?.message?.content || ''
   538→            } catch {
   539→              llmContent = ''
   540→            }
   541→            if (llmContent) send({ type: 'token', content: llmContent })
   542→            else send({ type: 'token', content: getFallback(intent, lang) })
   543→          }
   544→        }
   545→
   546→        // If streaming succeeded but produced no content, send fallback
   547→        if (!llmContent) {
   548→          // Already handled in the catch block or streaming produced nothing
   549→        }
   550→      } catch {
   551→        // Top-level error (e.g. OpenAI client init failure) — send fallback immediately
   552→        send({ type: 'token', content: getFallback(intent, lang) })
   553→      }
   554→
   555→      // Send structured data
   556→      if (products.length > 0) send({ type: 'products', products })
   557→      send({ type: 'suggestions', suggestions })
   558→      send({ type: 'done', whatsappUrl, detectedLang: lang })
   559→      controller.close()
   560→    },
   561→  })
   562→
   563→  return new Response(stream, { headers: { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', 'Connection': 'keep-alive' } })
   564→}
