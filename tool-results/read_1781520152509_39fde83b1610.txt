   142→  if (p.basePriceBDT === 'Inbox Price' || p.basePriceBDT === 'Low Price') return '  💰 সেরা দামের জন্য আমাদের সাথে যোগাযোগ করুন!'

   171→function buildProductContext(intent: Intent, userMsg: string): string {
   172→  const parts: string[] = []
   173→  const addProduct = (p: Product) => {
   174→    parts.push(`Name: ${safeName(p)} | Category: ${p.category.name}`)
   175→    parts.push(`Price Options:\n${parsePriceOptionsDetailed(p)}`)
   176→    parts.push(`Cheapest: ${getCheapestPlan(p)} | Stock: ${p.stockStatus} | Warranty: ${p.warranty || 'Full warranty'} | Delivery: ${p.deliveryTime}`)
   177→  }
   178→
   179→  switch (intent) {
   180→    case 'pin_inquiry': case 'greeting': case 'thanks': case 'goodbye':
   181→      parts.push('=== CATALOG OVERVIEW ===', briefSummary()); break
   182→    case 'specific_product': {
   183→      const p = findSpecificProduct(userMsg)
   184→      if (p) { parts.push('=== SPECIFIC PRODUCT ==='); addProduct(p); const f = parseFeatures(p); if (f.length) parts.push(`Features: ${f.slice(0, 8).join(', ')}`); const rel = findRelatedProducts(p.id, 3); if (rel.length) { parts.push('\n=== RELATED ==='); rel.forEach((r) => parts.push(`• ${safeName(r)} — From ${getCheapestPlan(r)}`)) } }
   185→      break
   186→    }
   187→    case 'category': {
   188→      const slug = detectCategorySlug(userMsg)
   189→      if (slug) { const { category, products, totalCount } = searchByCategory(slug); if (category && products.length) { parts.push(`=== CATEGORY: ${category.isAdult ? sanitizeText(category.name) : category.name} (${totalCount}) ===`); products.slice(0, 8).forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)}${p.isBestSeller ? ' 🔥' : p.isFeatured ? ' ⭐' : ''} — From ${getCheapestPlan(p)}`)) } }
   190→      break
   191→    }
   192→    case 'price_inquiry': case 'search': {
   193→      const results = searchProducts(userMsg, 6)
   194→      if (results.length) { parts.push('=== SEARCH RESULTS ==='); results.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} — From ${getCheapestPlan(p)} | ${p.category.name} | ${p.stockStatus}`)) }
   195→      const spec = findSpecificProduct(userMsg)
   196→      if (spec) { parts.push('\n=== EXACT MATCH ==='); addProduct(spec) }
   197→      break
   198→    }
   199→    case 'featured': {
   200→      const { products: feat } = getFeaturedProducts(6)
   201→      if (feat.length) { parts.push('=== FEATURED / BEST SELLERS ==='); feat.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} ${p.isBestSeller ? '🔥' : p.isNewArrival ? '✨' : ''} — From ${getCheapestPlan(p)}`)) }
   202→      break
   203→    }
   204→    case 'all_products': {
   205→      parts.push('=== FULL CATALOG ==='); getCatalogSummary().forEach((c) => parts.push(`📂 ${c.isAdult ? sanitizeText(c.name) : c.name} (${c.productCount}) — slug: ${c.slug}`, `   Popular: ${c.sampleProducts.slice(0, 3).join(', ')}`)); break
   206→    }
   207→    case 'comparison': {
   208→      const results = searchProducts(userMsg, 4)
   209→      if (results.length) { parts.push('=== COMPARISON DATA ==='); results.forEach((p) => { parts.push(`\n${safeName(p)}:`); parts.push(`  Price:\n${parsePriceOptionsDetailed(p)}`); parts.push(`  Cheapest: ${getCheapestPlan(p)}`); const f = parseFeatures(p); if (f.length) parts.push(`  Features: ${f.slice(0, 6).join(', ')}`); parts.push(`  Stock: ${p.stockStatus} | Warranty: ${p.warranty || 'Full'}`) }) }
   210→      break
   211→    }
   212→    case 'order_payment': case 'how_to_use': case 'warranty_delivery': {
   213→      const p = findSpecificProduct(userMsg)
   214→      if (p) { parts.push('=== RELEVANT PRODUCT ==='); addProduct(p) } else { parts.push('=== CATALOG OVERVIEW ===', briefSummary()) }
   215→      break
   216→    }
   217→    case 'out_of_scope': {
   218→      const { products: feat } = getFeaturedProducts(3)
   219→      if (feat.length) { parts.push('=== TOP PICKS ==='); feat.forEach((p, i) => parts.push(`${i + 1}. ${safeName(p)} — From ${getCheapestPlan(p)}`)) }
   220→      break
   221→    }
   222→  }
   223→  return parts.join('\n')
   224→}

   228→function collectProducts(intent: Intent, userMsg: string, hasRestrictedAccess: boolean): ProductCard[] {
   229→  const products: Product[] = []
   230→  switch (intent) {
   231→    case 'specific_product': { const p = findSpecificProduct(userMsg); if (p) { products.push(p); products.push(...findRelatedProducts(p.id, 3)) } break }
   232→    case 'category': { const slug = detectCategorySlug(userMsg); if (slug) products.push(...searchByCategory(slug).products.slice(0, 8)); break }

   364→// ─── POST Handler (SSE Streaming with OpenAI) ──────────────────────────────
   365→
   366→export async function POST(request: NextRequest) {
   367→  let body: { message?: string; history?: ChatMessage[]; sessionId?: string }
   368→  try { body = await request.json() } catch { return new Response(JSON.stringify({ error: 'Invalid JSON' }), { status: 400 }) }
   369→  const { message, history = [], sessionId = 'default' } = body
   370→  if (!message || typeof message !== 'string') return new Response(JSON.stringify({ error: 'Message required' }), { status: 400 })
   371→  if (!checkRateLimit(sessionId)) return new Response(JSON.stringify({ error: 'Rate limit exceeded' }), { status: 429 })
   372→
   373→  const intent = detectIntent(message, history)
   374→  const lang = detectLanguage(message)
   375→  const restrictedAccess = hasRestrictedAccessFromRequest(request)
   376→  const products = collectProducts(intent, message, restrictedAccess)
   377→  const suggestions = generateSuggestions(intent, lang)
   378→  const whatsappUrl = buildWhatsAppUrl({})
   379→  const systemPrompt = buildSystemPrompt(intent, lang, message)
   380→
   381→  // Sanitize history: only allow system/user/assistant roles, latest 8-10 messages, remove empty content
   382→  const cleanHistory = sanitizeHistory(history)
   383→  const messages: OpenAI.ChatCompletionMessageParam[] = [
   384→    { role: 'system', content: systemPrompt },
   385→    ...cleanHistory.map((m) => ({ role: m.role as 'user' | 'assistant', content: m.content })),
   386→    { role: 'user', content: message },
   387→  ]
   388→
   389→  const stream = new ReadableStream({
   390→    async start(controller) {
   391→      const enc = new TextEncoder()
   392→      const send = (data: Record<string, unknown>) => controller.enqueue(enc.encode(`data: ${JSON.stringify(data)}\n\n`))
   393→
   394→      // All LLM calls are wrapped with an 8-second timeout to prevent
   395→      // long hangs when the external API is unreachable.
   396→      const LLM_TIMEOUT_MS = 8000
   397→      let llmContent = ''
   398→
   399→      try {
   400→        // Try streaming with OpenAI (with timeout)
   401→        try {
   402→          const completion = await withTimeout(
   403→            getOpenAI().chat.completions.create({
   404→              model: 'gpt-4o-mini',
   405→              messages,
   406→              stream: true,
   407→            }),
   408→            LLM_TIMEOUT_MS,
   409→          )
   410→
   411→          // Use for-await-of loop over the async iterable stream
   412→          for await (const chunk of completion) {
   413→            const delta = chunk.choices?.[0]?.delta?.content
   414→            if (delta) {
   415→              send({ type: 'token', content: delta })
   416→              llmContent += delta
   417→            }
   418→          }
   419→        } catch (streamErr) {
   420→          // If streaming timed out, the API is unreachable — skip non-streaming attempt
   421→          // and go straight to fallback (avoids another wait)
   422→          if (streamErr instanceof Error && streamErr.message === 'LLM_TIMEOUT') {
   423→            send({ type: 'token', content: getFallback(intent, lang) })
   424→          } else {
   425→            // Non-timeout error — try non-streaming as fallback
   426→            try {
   427→              const completion = await withTimeout(
   428→                getOpenAI().chat.completions.create({
   429→                  model: 'gpt-4o-mini',
   430→                  messages,
   431→                  stream: false,
   432→                }),
   433→                LLM_TIMEOUT_MS,
   434→              )
   435→              llmContent = completion?.choices?.[0]?.message?.content || ''
   436→            } catch {
   437→              llmContent = ''
   438→            }
   439→            if (llmContent) send({ type: 'token', content: llmContent })
   440→            else send({ type: 'token', content: getFallback(intent, lang) })
   441→          }
   442→        }
   443→
   444→        // If streaming succeeded but produced no content, send fallback
   445→        if (!llmContent) {
   446→          // Already handled in the catch block or streaming produced nothing
   447→        }
   448→      } catch {
   449→        // Top-level error (e.g. OpenAI client init failure) — send fallback immediately
   450→        send({ type: 'token', content: getFallback(intent, lang) })
   451→      }
   452→
   453→      // Send structured data
   454→      if (products.length > 0) send({ type: 'products', products })
   455→      send({ type: 'suggestions', suggestions })
   456→      send({ type: 'done', whatsappUrl, detectedLang: lang })
   457→      controller.close()
   458→    },
   459→  })
   460→
   461→  return new Response(stream, { headers: { 'Content-Type': 'text/event-stream', 'Cache-Control': 'no-cache', 'Connection': 'keep-alive' } })
   462→}
