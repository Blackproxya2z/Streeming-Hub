   103→   103→function getCheapestPrice(product: ProductCard): string {
   104→   104→  try {
   105→   105→    const opts = JSON.parse(product.priceOptions || '[]')
   106→   106→    if (Array.isArray(opts) && opts.length > 0) {
   107→   107→      const sorted = [...opts].sort((a: Record<string, unknown>, b: Record<string, unknown>) => {
   108→   108→        const priceA = parseInt(String(a.priceBDT).replace(/\D/g, '') || '0')
   109→   109→        const priceB = parseInt(String(b.priceBDT).replace(/\D/g, '') || '0')
   110→   110→        return priceA - priceB
   111→   111→      })
   112→   112→      return String(sorted[0].priceBDT)
   113→   113→    }
   114→   114→  } catch {
   115→   115→    // fallback to base price
   116→   116→  }
   117→   117→  return product.basePriceBDT
   118→   118→}

   122→   122→function detectLanguageClient(message: string): DetectedLang {
   123→   123→  if (/[\u0980-\u09FF]/.test(message)) return 'bangla'
   124→   124→  const patterns = [/koto/i, /taka/i, /lagbe/i, /chai/i, /order\s*korbo/i, /nite\s*chai/i, /kinte\s*chai/i, /dekhao/i, /ki\s*ki/i, /sob/i, /nam/i, /amar/i, /apnar/i, /bhai/i, /kichu/i, /kemon/i, /valo/i, /ase/i, /ache/i, /korbo/i, /parbo/i, /kivabe/i, /keno/i]
   125→   125→  if (patterns.filter((p) => p.test(message)).length >= 2) return 'banglish'
   126→   126→  return 'english'
   127→   127→}

   131→   131→// Chrome loads voices asynchronously — this helper ensures they're available
   132→   132→function getVoicesAsync(): Promise<SpeechSynthesisVoice[]> {
   133→   133→  return new Promise((resolve) => {
   134→   134→    if (typeof window === 'undefined' || !window.speechSynthesis) {
   135→   135→      resolve([])
   136→   136→      return
   137→   137→    }
   138→   138→    const voices = window.speechSynthesis.getVoices()
   139→   139→    if (voices.length > 0) {
   140→   140→      resolve(voices)
   141→   141→      return
   142→   142→    }
   143→   143→    // Voices not loaded yet — wait for voiceschanged event
   144→   144→    const onVoicesChanged = () => {
   145→   145→      const v = window.speechSynthesis.getVoices()
   146→   146→      if (v.length > 0) {
   147→   147→        window.speechSynthesis.removeEventListener('voiceschanged', onVoicesChanged)
   148→   148→        resolve(v)
   149→   149→      }
   150→   150→    }
   151→   151→    window.speechSynthesis.addEventListener('voiceschanged', onVoicesChanged)
   152→   152→    // Safety: resolve after 2s even if no voices loaded
   153→   153→    setTimeout(() => {
   154→   154→      window.speechSynthesis.removeEventListener('voiceschanged', onVoicesChanged)
   155→   155→      resolve(window.speechSynthesis.getVoices())
   156→   156→    }, 2000)
   157→   157→  })
   158→   158→}

   184→   184→async function speakWithServerTTS(text: string, lang: DetectedLang, onEnd?: () => void): Promise<boolean> {
   185→   185→  // Clean text for TTS on the client side too (for the request body)
   186→   186→  const cleanText = text
   187→   187→    .replace(/[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F1E0}-\u{1F1FF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}\u{FE00}-\u{FE0F}\u{1F900}-\u{1F9FF}\u{200D}\u{20E3}\u{E0020}-\u{E007F}]/gu, '')
   188→   188→    .replace(/\*\*(.+?)\*\*/g, '$1')
   189→   189→    .replace(/\[(.+?)\]\((.+?)\)/g, '$1')
   190→   190→    .replace(/[•\-]\s/g, '')
   191→   191→    .replace(/\d+[.)]\s/g, '')
   192→   192→    .replace(/৳/g, 'টাকা')
   193→   193→    .replace(/BDT/gi, 'টাকা')
   194→   194→    .replace(/\bbKash\b/gi, 'বিকাশ')
   195→   195→    .replace(/\bNagad\b/gi, 'নগদ')
   196→   196→    .replace(/\bWhatsApp\b/gi, 'হোয়াটসঅ্যাপ')
   197→   197→    .replace(/↵/g, ' ')
   198→   198→    .replace(/\n/g, ' ')
   199→   199→    .trim()
   200→   200→
   201→   201→  if (!cleanText) { onEnd?.(); return true }
   202→   202→
   203→   203→  try {
   204→   204→    // Stop any currently playing audio
   205→   205→    if (currentAudio) {
   206→   206→      currentAudio.pause()
   207→   207→      currentAudio = null
   208→   208→    }
   209→   209→
   210→   210→    const res = await fetch('/api/tts', {
   211→   211→      method: 'POST',
   212→   212→      headers: { 'Content-Type': 'application/json' },
   213→   213→      body: JSON.stringify({ text: cleanText, lang }),
   214→   214→    })
   215→   215→
   216→   216→    // If server TTS returns 503 with fallback flag, use browser TTS
   217→   217→    if (res.status === 503) {
   218→   218→      console.warn('Server TTS unavailable (ElevenLabs + OpenAI failed), falling back to browser TTS')
   219→   219→      return false
   220→   220→    }
   221→   221→
   222→   222→    if (!res.ok) {
   223→   223→      console.warn('Server TTS failed, falling back to browser TTS')
   224→   224→      return false
   225→   225→    }
   226→   226→
   227→   227→    const audioBlob = await res.blob()
   228→   228→    const audioUrl = URL.createObjectURL(audioBlob)
   229→   229→    const audio = new Audio(audioUrl)
   230→   230→    currentAudio = audio
   231→   231→
   232→   232→    return new Promise<boolean>((resolve) => {
   233→   233→      const safetyTimer = setTimeout(() => {
   234→   234→        audio.pause()
   235→   235→        currentAudio = null
   236→   236→        URL.revokeObjectURL(audioUrl)
   237→   237→        onEnd?.()
   238→   238→        resolve(true)
   239→   239→      }, 45000)
   240→   240→
   241→   241→      audio.onended = () => {
   242→   242→        clearTimeout(safetyTimer)
   243→   243→        currentAudio = null
   244→   244→        URL.revokeObjectURL(audioUrl)
   245→   245→        onEnd?.()
   246→   246→        resolve(true)
   247→   247→      }
   248→   248→
   249→   249→      audio.onerror = () => {
   250→   250→        clearTimeout(safetyTimer)
   251→   251→        currentAudio = null
   252→   252→        URL.revokeObjectURL(audioUrl)
   253→   253→        console.warn('Audio playback error, falling back to browser TTS')
   254→   254→        resolve(false)
   255→   255→      }
   256→   256→
   257→   257→      audio.play().catch(() => {
   258→   258→        clearTimeout(safetyTimer)
   259→   259→        currentAudio = null
   260→   260→        URL.revokeObjectURL(audioUrl)
   261→   261→        resolve(false)
   262→   262→      })
   263→   263→    })
   264→   264→  } catch {
   265→   265→    console.warn('Server TTS error, falling back to browser TTS')
   266→   266→    return false
   267→   267→  }
   268→   268→}

   270→   270→function stopServerAudio() {
   271→   271→  if (currentAudio) {
   272→   272→    currentAudio.pause()
   273→   273→    currentAudio = null
   274→   274→  }

   277→   277→async function speak(text: string, lang: DetectedLang, onEnd?: () => void): Promise<SpeechSynthesisUtterance | null> {
   278→   278→  if (typeof window === 'undefined') return null
   279→   279→
   280→   280→  // Stop any existing audio
   281→   281→  stopServerAudio()
   282→   282→  if (window.speechSynthesis) window.speechSynthesis.cancel()
   283→   283→  stopChromeKeepAlive()
   284→   284→
   285→   285→  // Try Server TTS first (ElevenLabs → OpenAI → Browser fallback)
   286→   286→  const serverSuccess = await speakWithServerTTS(text, lang, onEnd)
   287→   287→  if (serverSuccess) return null // Server TTS handled it
   288→   288→
   289→   289→  // Fallback to browser TTS
   290→   290→  if (!window.speechSynthesis) { onEnd?.(); return null }
   291→   291→
   292→   292→  // Clean text for speech
   293→   293→  const cleanText = text
   294→   294→    .replace(/[\u{1F600}-\u{1F64F}\u{1F300}-\u{1F5FF}\u{1F680}-\u{1F6FF}\u{1F1E0}-\u{1F1FF}\u{2600}-\u{26FF}\u{2700}-\u{27BF}\u{FE00}-\u{FE0F}\u{1F900}-\u{1F9FF}\u{200D}\u{20E3}\u{E0020}-\u{E007F}]/gu, '')
   295→   295→    .replace(/\*\*(.+?)\*\*/g, '$1')
   296→   296→    .replace(/\[(.+?)\]\((.+?)\)/g, '$1')
   297→   297→    .replace(/[•\-]\s/g, '')
   298→   298→    .replace(/\d+[.)]\s/g, '')
   299→   299→    .replace(/৳/g, 'টাকা')
   300→   300→    .replace(/BDT/gi, 'টাকা')
   301→   301→    .replace(/\bbKash\b/gi, 'বিকাশ')
   302→   302→    .replace(/\bNagad\b/gi, 'নগদ')
   303→   303→    .replace(/\bWhatsApp\b/gi, 'হোয়াটসঅ্যাপ')
   304→   304→    .replace(/↵/g, ' ')
   305→   305→    .replace(/\n/g, ' ')
   306→   306→    .trim()
   307→   307→
   308→   308→  if (!cleanText) { onEnd?.(); return null }
   309→   309→
   310→   310→  const voices = await getVoicesAsync()
   311→   311→  const utterance = new SpeechSynthesisUtterance(cleanText)
   312→   312→
   313→   313→  if (lang === 'bangla' || lang === 'banglish') {
   314→   314→    const bnVoice = voices.find(v => v.lang === 'bn-BD') || voices.find(v => v.lang === 'bn-IN')
   315→   315→    if (bnVoice) {
   316→   316→      utterance.voice = bnVoice
   317→   317→      utterance.lang = bnVoice.lang
   318→   318→    } else {
   319→   319→      const hiVoice = voices.find(v => v.lang === 'hi-IN')
   320→   320→      if (hiVoice) { utterance.voice = hiVoice; utterance.lang = 'hi-IN' }
   321→   321→      else { utterance.lang = 'bn-BD' }
   322→   322→    }
   323→   323→  } else {
   324→   324→    utterance.lang = 'en-US'
   325→   325→    const enVoice = voices.find(v => v.lang === 'en-US' && v.name.toLowerCase().includes('female'))
   326→   326→      || voices.find(v => v.lang === 'en-US')
   327→   327→    if (enVoice) utterance.voice = enVoice
   328→   328→  }
   329→   329→
   330→   330→  utterance.rate = 0.9
   331→   331→  utterance.pitch = 1.15
   332→   332→  utterance.volume = 1.0
   333→   333→
   334→   334→  startChromeKeepAlive()
   335→   335→
   336→   336→  const safetyTimer = setTimeout(() => {
   337→   337→    if (window.speechSynthesis?.speaking) window.speechSynthesis.cancel()
   338→   338→    stopChromeKeepAlive()
   339→   339→    onEnd?.()
   340→   340→  }, 45000)
   341→   341→
   342→   342→  const wrappedOnEnd = () => {
   343→   343→    clearTimeout(safetyTimer)
   344→   344→    stopChromeKeepAlive()
   345→   345→    onEnd?.()
   346→   346→  }
   347→   347→
   348→   348→  utterance.onend = wrappedOnEnd
   349→   349→  utterance.onerror = () => wrappedOnEnd()
   350→   350→
   351→   351→  await new Promise(resolve => setTimeout(resolve, 100))
   352→   352→  window.speechSynthesis.speak(utterance)
   353→   353→  return utterance
   354→   354→}

   358→   358→function formatMessage(text: string): ReactNode[] {
   359→   359→  const lines = text.split('\n')
   360→   360→  const elements: ReactNode[] = []
   361→   361→
   362→   362→  lines.forEach((line, lineIdx) => {

   366→   366→      elements.push(
   367→   367→        <span key={`line-${lineIdx}`}>
   368→   368→          {processedLine}
   369→   369→          <br />
   370→   370→        </span>
   371→   371→      )
   372→   372→    } else {
   373→   373→      elements.push(<span key={`line-${lineIdx}`}>{processedLine}</span>)
   374→   374→    }
   375→   375→  })
   376→   376→
   377→   377→  return elements
   378→   378→}
   379→   379→
   380→   380→function processInlineFormatting(line: string): ReactNode[] {
   381→   381→  const elements: ReactNode[] = []
   382→   382→  let remaining = line
   383→   383→
   384→   384→  // Check if this is a bullet point line (• or -)
   385→   385→  const bulletMatch = remaining.match(/^(\s*)([•\-])\s(.*)/)
   386→   386→  if (bulletMatch) {
   387→   387→    const [, indent, , content] = bulletMatch
   388→   388→    elements.push(
   389→   389→      <span key={`bullet-${indent.length}`} className="flex items-start gap-1.5 my-0.5">
   390→   390→        <span className="mt-[7px] shrink-0 h-1.5 w-1.5 rounded-full bg-primary" />
   391→   391→        <span>{processInlineFormatting(content)}</span>
   392→   392→      </span>
   393→   393→    )
   394→   394→    return elements
   395→   395→  }
   396→   396→
   397→   397→  // Check if this is a numbered list item (1. 2. etc.)
   398→   398→  const numberedMatch = remaining.match(/^(\s*)(\d+)[.)]\s(.*)/)
   399→   399→  if (numberedMatch) {
   400→   400→    const [, , num, content] = numberedMatch
   401→   401→    elements.push(
   402→   402→      <span key={`num-${num}`} className="flex items-start gap-1.5 my-0.5">
   403→   403→        <span className="shrink-0 min-w-[18px] h-[18px] rounded-full bg-primary/15 text-primary flex items-center justify-center text-[10px] font-bold leading-none mt-px">
   404→   404→          {num}
   405→   405→        </span>
   406→   406→        <span>{processInlineFormatting(content)}</span>
   407→   407→      </span>
   408→   408→    )
   409→   409→    return elements
   410→   410→  }
   411→   411→
   412→   412→  // Process inline: **bold** and links
   413→   413→  let keyIdx = 0
   414→   414→  while (remaining.length > 0) {
   415→   415→    // Bold pattern: **text**
   416→   416→    const boldMatch = remaining.match(/\*\*(.+?)\*\*/)
   417→   417→    if (boldMatch && boldMatch.index !== undefined) {
   418→   418→      if (boldMatch.index > 0) {
   419→   419→        elements.push(<span key={`t-${keyIdx++}`}>{remaining.slice(0, boldMatch.index)}</span>)
   420→   420→      }
   421→   421→      elements.push(
   422→   422→        <strong key={`b-${keyIdx++}`} className="font-semibold text-teal-700">
   423→   423→          {boldMatch[1]}
   424→   424→        </strong>
